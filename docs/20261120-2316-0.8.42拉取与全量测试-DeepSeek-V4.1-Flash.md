```bash

# 浏览器（aarch64）：npmmirror 的 headless shell 1155 arm64 包，软链到 1243 目录
D=/root/.cache/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-linux-arm64

         ui-ability-flags ui-flow-timeline ui-summon ui-elite ui-layout \
```
